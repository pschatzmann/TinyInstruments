131 size_t render(int16_t* out, size_t frames) {
132 processEvents();
133 float left[TI_BLOCK_SIZE];
134 float right[TI_BLOCK_SIZE];
135 size_t done = 0;
136 while (done < frames) {
137 int n = frames - done > TI_BLOCK_SIZE ? TI_BLOCK_SIZE : frames - done;
138 renderBlock(left, right, n);
139 int16_t* dest = out + done * 2;
140 const float gain = master * 32767.0f;
141 for (int j = 0; j < n; j++) {
142 dest[2 * j] = toInt16(left[j] * gain);
143 dest[2 * j + 1] = toInt16(right[j] * gain);
144 }
145 done += n;
146 }
147 return frames;
148 }

224 void doProgramChange(uint8_t ch, uint8_t program) {
225 if (p_bank == nullptr) return;
226 const InstrumentBank::Entry* entry = p_bank->find(ch, program);
227 if (entry == nullptr) return;
228 ChannelState& cs = channels[ch];
229 // monophonic state and palm mute belong to the old instrument
230 cs.held_count = 0;
231 cs.mute = false;
232 // the notes of the old instrument fade out (an effect could change)
233 for (int j = 0; j < VOICE_COUNT; j++) {
234 SamplerVoice& v = voices[j];
235 if (v.isActive() && v.channel == ch && cs.effect != entry->effect) {
236 v.release(0.05f * sample_rate);
237 v.fading = true;
238 v.key_down = false;
239 }
240 }
241 cs.instrument = entry->instrument;
242 cs.effect = entry->effect;
243 cs.program = program;
244 }

252 void processEvents() {
253 SynthEvent e;
254 while (queue.pop(e)) {
255 switch (e.type) {
256 case SynthEvent::NoteOn:
257 doNoteOn(e.channel, e.a, e.b);
258 break;
259 case SynthEvent::NoteOff:
260 doNoteOff(e.channel, e.a);
261 break;
262 case SynthEvent::ControlChange:
263 doControlChange(e.channel, e.a, e.b);
264 break;
265 case SynthEvent::PitchBend:
266 doPitchBend(e.channel, e.value);
267 break;
268 case SynthEvent::ProgramChange:
269 doProgramChange(e.channel, e.a);
270 break;
271 default:
272 break;
273 }
274 }

277 void renderBlock(float* left, float* right, int n) {
278 memset(left, 0, n * sizeof(float));
279 memset(right, 0, n * sizeof(float));
280 // channels without effect: directly into the mix
281 for (int j = 0; j < VOICE_COUNT; j++) {
282 if (voices[j].isActive() && channels[voices[j].channel].effect == nullptr)
283 voices[j].render(left, right, n);
284 }
285 // each effect processes the voices of its channels
286 for (int ch = 0; ch < TI_MIDI_CHANNELS; ch++) {
287 ChannelEffect* effect = channels[ch].effect;
288 if (effect == nullptr) continue;
289 bool done = false; // effect shared with a previous channel
290 for (int prev = 0; prev < ch; prev++)
291 if (channels[prev].effect == effect) done = true;
292 if (done) continue;
293 float fx_left[TI_BLOCK_SIZE];
294 float fx_right[TI_BLOCK_SIZE];
295 memset(fx_left, 0, n * sizeof(float));
296 memset(fx_right, 0, n * sizeof(float));
297 bool active = effect->hasTail();
298 for (int j = 0; j < VOICE_COUNT; j++) {
299 SamplerVoice& v = voices[j];
300 if (v.isActive() && channels[v.channel].effect == effect) {
301 v.render(fx_left, fx_right, n);
302 active = true;
303 }
304 }
305 if (!active) continue;
306 effect->setup(sample_rate);
307 effect->process(fx_left, fx_right, n);
308 for (int j = 0; j < n; j++) {
309 left[j] += fx_left[j];
310 right[j] += fx_right[j];
311 }
312 }

317 static inline int16_t toInt16(float x) {
318 const float knee = 0.9f * 32767.0f;
319 if (x > knee || x < -knee) {
320 float a = fabsf(x);
321 float over = (a - knee) * (10.0f / 32767.0f);
322 a = knee + 0.1f * 32767.0f * over / (1.0f + over);
323 x = x < 0 ? -a : a;
324 }
325 return (int16_t)x;
326 }

328 void doNoteOn(uint8_t ch, uint8_t note, uint8_t velocity) {
329 ChannelState& cs = channels[ch];
330 const Instrument* inst = cs.instrument;
331 if (inst == nullptr || (inst->samples == nullptr && inst->organ == nullptr)) return;
332 if (inst->monophonic) {
333 bool legato = inst->legato && cs.held_count > 0;
334 removeHeld(cs, note);
335 if (cs.held_count < 16) {
336 cs.held_notes[cs.held_count] = note;
337 cs.held_velocity[cs.held_count] = velocity;
338 cs.held_count++;
339 }
340 playMono(ch, note, velocity, legato);
341 return;
342 }
343 startNote(ch, note, velocity, false);
344 }

347 void playMono(uint8_t ch, uint8_t note, uint8_t velocity, bool legato) {
348 const Instrument* inst = channels[ch].instrument;
349 float fade = (legato ? inst->legatoTime : inst->restrikeTime) * sample_rate;
350 for (int j = 0; j < VOICE_COUNT; j++) {
351 SamplerVoice& v = voices[j];
352 if (v.isActive() && !v.fading && v.channel == ch) {
353 v.release(fade);
354 v.key_down = false;
355 v.sustained = false;
356 v.fading = true;
357 }
358 }
359 startNote(ch, note, velocity, legato);
360 }

362 void removeHeld(ChannelState& cs, uint8_t note) {
363 int out = 0;
364 for (int j = 0; j < cs.held_count; j++) {
365 if (cs.held_notes[j] != note) {
366 cs.held_notes[out] = cs.held_notes[j];
367 cs.held_velocity[out] = cs.held_velocity[j];
368 out++;
369 }
370 }
371 cs.held_count = out;
372 }

375 void startNote(uint8_t ch, uint8_t key, uint8_t velocity, bool legato) {
376 ChannelState& cs = channels[ch];
377 const Instrument* inst = cs.instrument;
378 if (inst->organ != nullptr) {
379 startOrganNote(ch, key, velocity);
380 return;
381 }
382 uint8_t note = soundingNote(cs, key);
383 const SampleZone* zone = inst->samples->findZone(note, velocity, inst->exactKeys);
384 if (zone == nullptr) return;
385
386 // choke group: e.g. a closed hi-hat silences an open one
387 if (zone->group != 0) {
388 for (int j = 0; j < VOICE_COUNT; j++) {
389 SamplerVoice& v = voices[j];
390 if (v.isActive() && v.channel == ch && v.zone()->offBy == zone->group) {
391 v.release(inst->chokeTime * sample_rate);
392 v.fading = true;
393 }
394 }
395 }
396
397 // a note which is played again: fade out the old one
398 for (int j = 0; j < VOICE_COUNT; j++) {
399 SamplerVoice& v = voices[j];
400 if (v.isActive() && !v.fading && v.channel == ch && v.key == key) {
401 v.release(inst->restrikeTime * sample_rate);
402 v.key_down = false;
403 v.sustained = false;
404 v.fading = true;
405 }
406 }
407
408 SamplerVoice& v = allocateVoice(ch, inst->maxNotes);
409 // rate conversion and transposition: the transposition is done by the
410 // resampling or with PSOLA (which keeps the length of the sound)
411 float rate_ratio = (float)inst->samples->sampleRate / (float)sample_rate;
412 float transpose = powf(2.0f, (note - zone->rootKey + zone->tune * 0.01f) / 12.0f);
413 bool psola = inst->pitchShift == PitchShift::Psola && PsolaSource::isSupported(zone) &&
414 fabsf(transpose - 1.0f) > 0.0001f;
415 float step = psola ? rate_ratio : rate_ratio * transpose;
416
417 // loudness: absolute velocity (the layers only change the timbre)
418 float gain = powf(velocity / 127.0f, inst->velocityCurve) * inst->volume;
419
420 // brightness: lowpass which opens with the velocity inside of the layer
421 float rel = (float)velocity / (float)zone->hiVel;
422 if (rel > 1.0f) rel = 1.0f;
423 float cutoff = inst->minCutoff + (20000.0f - inst->minCutoff) * rel * rel;
424 bool muted = inst->muteController != 0 && cs.mute;
425 if (muted && cutoff > inst->muteCutoff) cutoff = inst->muteCutoff;
426 float lowpass = 1.0f;
427 if (cutoff < 0.45f * sample_rate) {
428 lowpass = 1.0f - expf(-6.2831853f * cutoff / sample_rate);
429 }
430
431 // stereo position of the key: constant power panning
432 float pos = ((float)note - 64.5f) / 44.0f * inst->stereoWidth;
433 if (pos < -1.0f) pos = -1.0f;
434 if (pos > 1.0f) pos = 1.0f;
435 float angle = (pos + 1.0f) * 0.78539816f;
436 float pan_left = cosf(angle) * 1.41421356f;
437 float pan_right = sinf(angle) * 1.41421356f;
438
439 v.start(zone, step, gain, lowpass, pan_left, pan_right, legato,
440 legato ? inst->legatoTime * sample_rate : 0.0f, psola ? transpose : 0.0f);
441 if (muted) {
442 // dB per second of the output -> ln per sample of the zone
443 v.setExtraDecay(-inst->muteDecay * 0.11512925f / (step * sample_rate));
444 }
445 v.setPitchFactor(cs.bend);
446 v.setChannelGain(channelGain(cs), true);
447 v.setVibrato(inst->vibratoRate, inst->vibratoDepth * cs.modulation, inst->autoVibratoDepth,
448 inst->autoVibratoDelay * sample_rate, inst->autoVibratoRise * sample_rate,
449 sample_rate);
450 v.channel = ch;
451 v.note = note;
452 v.key = key;
453 v.velocity = velocity;
454 v.key_down = true;
455 v.sustained = false;
456 v.fading = false;
457 v.age = ++note_counter;
458 }

461 void startOrganNote(uint8_t ch, uint8_t key, uint8_t velocity) {
462 ChannelState& cs = channels[ch];
463 const Instrument* inst = cs.instrument;
464 uint8_t note = soundingNote(cs, key);
465 // single triggered percussion: only if no other key of the channel is held
466 bool percussion = true;
467 for (int j = 0; j < VOICE_COUNT; j++) {
468 SamplerVoice& v = voices[j];
469 if (v.isActive() && v.channel == ch && !v.fading && (v.key_down || v.sustained)) percussion = false;
470 // a note which is played again: fade out the old one
471 if (v.isActive() && !v.fading && v.channel == ch && v.key == key) {
472 v.release(inst->restrikeTime * sample_rate);
473 v.key_down = false;
474 v.sustained = false;
475 v.fading = true;
476 }
477 }
478 SamplerVoice& v = allocateVoice(ch, inst->maxNotes);
479 float gain = powf(velocity / 127.0f, inst->velocityCurve) * inst->volume;
480 float pos = ((float)note - 64.5f) / 44.0f * inst->stereoWidth;
481 if (pos < -1.0f) pos = -1.0f;
482 if (pos > 1.0f) pos = 1.0f;
483 float angle = (pos + 1.0f) * 0.78539816f;
484 v.startOrgan(inst->organ, note, (float)sample_rate, percussion, gain, cosf(angle) * 1.41421356f,
485 sinf(angle) * 1.41421356f);
486 v.setPitchFactor(cs.bend);
487 v.setChannelGain(channelGain(cs), true);
488 v.setVibrato(inst->vibratoRate, inst->vibratoDepth * cs.modulation, 0.0f, 0.0f, 1.0f, sample_rate);
489 v.channel = ch;
490 v.note = note;
491 v.key = key;
492 v.velocity = velocity;
493 v.key_down = true;
494 v.sustained = false;
495 v.fading = false;
496 v.age = ++note_counter;
497 }

506 void doNoteOff(uint8_t ch, uint8_t note) {
507 ChannelState& cs = channels[ch];
508 const Instrument* inst = cs.instrument;
509 if (inst != nullptr && inst->monophonic) {
510 bool was_sounding = cs.held_count > 0 && cs.held_notes[cs.held_count - 1] == note;
511 removeHeld(cs, note);
512 // return to the last key which is still held (e.g. trill)
513 if (was_sounding && cs.held_count > 0 && !cs.sustain) {
514 uint8_t last = cs.held_count - 1;
515 playMono(ch, cs.held_notes[last], cs.held_velocity[last], inst->legato);
516 return;
517 }
518 }
519 for (int j = 0; j < VOICE_COUNT; j++) {
520 SamplerVoice& v = voices[j];
521 if (!v.isActive() || !v.key_down || v.channel != ch || v.key != note) continue;
522 v.key_down = false;
523 if (cs.sustain) {
524 v.sustained = true;
525 } else {
526 releaseVoice(v, cs);
527 }
528 }

542 void doControlChange(uint8_t ch, uint8_t controller, uint8_t value) {
543 ChannelState& cs = channels[ch];
544 if (cs.effect != nullptr) cs.effect->controlChange(controller, value);
545 if (cs.instrument != nullptr && cs.instrument->organ != nullptr)
546 cs.instrument->organ->controlChange(controller, value);
547 if (cs.instrument != nullptr && controller == cs.instrument->muteController &&
548 controller != 0) {
549 cs.mute = value >= 64;
550 return;
551 }
552 switch (controller) {
553 case 1: // modulation wheel: vibrato depth (0 for the pianos and drums)
554 cs.modulation = value / 127.0f;
555 for (int j = 0; j < VOICE_COUNT; j++) {
556 SamplerVoice& v = voices[j];
557 if (v.isActive() && v.channel == ch && cs.instrument != nullptr)
558 v.setVibratoDepth(cs.instrument->vibratoDepth * cs.modulation);
559 }
560 break;
561 case 2: // breath controller
562 cs.breath = value / 127.0f;
563 break;
564 case 7:
565 cs.volume = value / 127.0f;
566 break;
567 case 11:
568 cs.expression = value / 127.0f;
569 break;
570 case 64:
571 setSustain(ch, value >= 64);
572 break;
573 case 120: // all sound off
574 for (int j = 0; j < VOICE_COUNT; j++)
575 if (voices[j].channel == ch) voices[j].stop();
576 break;
577 case 121: // reset all controllers
578 cs.expression = 1.0f;
579 cs.breath = 1.0f;
580 cs.modulation = 0.0f;
581 setSustain(ch, false);
582 doPitchBend(ch, 8192);
583 break;
584 case 123: // all notes off
585 setSustain(ch, false);
586 cs.held_count = 0;
587 for (int j = 0; j < VOICE_COUNT; j++) {
588 SamplerVoice& v = voices[j];
589 if (v.isActive() && v.channel == ch) {
590 v.key_down = false;
591 v.release(0.1f * sample_rate);
592 }
593 }
594 break;
595 default:
596 break;
597 }
598 updateChannelGain(ch);
599 }

644 void stealNote(int ch) {
645 int victim = -1;
646 float best_level = 2.0f;
647 for (int j = 0; j < VOICE_COUNT; j++) {
648 const SamplerVoice& v = voices[j];
649 if (!v.isActive() || v.fading || (ch >= 0 && v.channel != ch)) continue;
650 if (!v.key_down && v.level() < best_level) {
651 best_level = v.level();
652 victim = j;
653 }
654 }
655 if (victim < 0) {
656 for (int j = 0; j < VOICE_COUNT; j++) {
657 const SamplerVoice& v = voices[j];
658 if (!v.isActive() || v.fading || (ch >= 0 && v.channel != ch)) continue;
659 if (victim < 0 || v.age < voices[victim].age) victim = j;
660 }
661 }
662 if (victim >= 0) {
663 voices[victim].release(steal_fade * sample_rate);
664 voices[victim].fading = true;
665 }
666 }

671 SamplerVoice& allocateVoice(uint8_t ch, int instrumentMaxNotes) {
672 if (activeNotes(ch) >= instrumentMaxNotes) {
673 stealNote(ch);
674 } else if (activeNotes() >= max_notes) {
675 stealNote(-1);
676 }
677 for (int j = 0; j < VOICE_COUNT; j++)
678 if (!voices[j].isActive()) return voices[j];
679 // all voices busy (only fading ones are left): take the quietest
680 int quietest = 0;
681 for (int j = 1; j < VOICE_COUNT; j++) {
682 bool better = voices[j].fading && (!voices[quietest].fading ||
683 voices[j].level() < voices[quietest].level());
684 if (better) quietest = j;
