72 loop_mark++;

109 inline float next() {
110 // start the grains which begin at this position
111 while (!done && pos >= next_start) startGrain();
112
113 float acc = 0.0f, weight = 0.0f;
114 if (raw_active) {
115 float w = 1.0f;
116 if (pos >= raw_fade_start) {
117 float x = (float)(pos - raw_fade_start) * raw_fade_inv;
118 if (x >= 1.0f) {
119 raw_active = false;
120 w = 0.0f;
121 } else {
122 w = 1.0f - smoothstep(x);
123 }
124 }
125 if (raw_active) {
126 decodeTo(pos);
127 acc += w * ring[pos & MASK];
128 weight += w;
129 }
130 }
131 // the grains (the ring buffer is already decoded up to their end)
132 for (int j = 0; j < grain_count;) {
133 Grain& g = grains[j];
134 float sm = smoothstep(g.x);
135 float w = g.rising ? sm : 1.0f - sm;
136 acc += w * ring[g.u & MASK];
137 weight += w;
138 g.u++;
139 g.x += g.inc;
140 if (--g.remaining <= 0) {
141 if (g.rising) {
142 // second half: falling window
143 g.rising = false;
144 g.x = 0.0f;
145 g.inc = g.inc_right;
146 g.remaining = g.right;
147 } else {
148 grains[j] = grains[--grain_count]; // finished: remove
149 continue;
150 }
151 }
152 j++;
153 }
154 pos++;
155 // normalized overlap add: constant level for any grain distance
156 return weight > 0.001f ? acc * (1.0f / 32768.0f) / weight : 0.0f;
157 }

217 inline void decodeTo(int32_t u) {
218 while (decoded <= u) {
219 int16_t v = 0;
220 if (decode_idx >= z->loopEnd && z->isLooped()) {
221 decode_idx = z->loopStart;
222 decoder.seek(z->loopStart, z->loopPredictor, z->loopIndex);
223 }
224 if (decode_idx < z->length) {
225 v = decoder.next();
226 decode_idx++;
227 }
228 ring[decoded & MASK] = v;
229 decoded++;
230 }

234 void startGrain() {
235 if (grain_count < MAX_GRAINS) {
236 Grain& g = grains[grain_count++];
237 int32_t left = leftOf(next_mark);
238 g.right = rightOf(next_mark);
239 // the grain may start a little later than planned (rounding)
240 int32_t late = pos - next_start;
241 late = late > 0 ? (late < left ? late : left - 1) : 0;
242 g.u = next_mark_u - left + late;
243 g.remaining = left - late;
244 g.inc = 1.0f / left;
245 g.inc_right = 1.0f / g.right;
246 g.x = late * g.inc;
247 g.rising = true;
248 // decode the whole grain: no checks are needed when reading it
249 decodeTo(next_mark_u + g.right);
250 }
251 // next synthesis mark: one period of the new pitch later (the period
252 // after the mark: for a ratio of 1 the marks are met exactly, also if the
253 // periods of the recording vary)
254 next_syn += rightOf(next_mark) / ratio;
255 if (!findMark(next_syn)) {
256 done = true; // end of the zone (no loop)
257 return;
258 }
259 next_start = (int32_t)(next_syn + 0.5) - leftOf(next_mark);
260 }

264 bool findMark(double t) {
265 // offset of the loop repetition which contains t
266 int32_t offset = 0;
267 double local = t;
268 if (loop_len > 0 && t >= z->loopEnd) {
269 int32_t rep = (int32_t)((t - z->loopStart) / loop_len);
270 offset = rep * loop_len;
271 local = t - (double)offset;
272 }
273 // the time jumped back into the loop: restart at the first loop mark
274 if ((double)cursor_pos > local + leftOf(cursor)) {
275 cursor = loop_mark;
276 cursor_pos = loop_mark_pos;
277 }
278 while (cursor + 1 < z->markCount &&
279 fabs((double)(cursor_pos + z->marks[cursor + 1]) - local) <= fabs((double)cursor_pos - local)) {
280 cursor++;
281 cursor_pos += z->marks[cursor];
282 }
283 if (loop_len == 0 && cursor + 1 >= z->markCount && local > cursor_pos + rightOf(cursor)) return false;
284 next_mark = cursor;
285 next_mark_u = cursor_pos + offset;
