271 void advanceEnvelopes(int frames) {
272 if (decay_active && decay_mul != 1.0f) decay_env *= blockPow(decay_mul, decay_mul_block, frames);
273 if (release_mul != 1.0f) release_env *= blockPow(release_mul, release_mul_block, frames);
274 if (attack_env < 1.0f) {
275 attack_env += attack_inc * frames;
276 if (attack_env > 1.0f) attack_env = 1.0f;
277 }
278 // smoothing of the channel gain: ~10 ms
279 float k = 0.002f * frames;
280 channel_gain += (k > 1.0f ? 1.0f : k) * (channel_gain_target - channel_gain);
281 }

284 static inline float blockPow(float mul, float& cache, int frames) {
285 if (frames != TI_BLOCK_SIZE) return powf(mul, frames);
286 if (cache < 0.0f) cache = powf(mul, TI_BLOCK_SIZE);

291 void updateVibrato(int frames) {
292 float depth = vib_depth;
293 if (auto_depth > 0.0f && elapsed > auto_delay) {
294 float amount = (elapsed - auto_delay) / auto_rise;
295 depth += auto_depth * (amount > 1.0f ? 1.0f : amount);
296 }
297 if (depth <= 0.0f) {
298 step = base_step * pitch_factor;
299 return;
300 }
301 lfo_phase += lfo_inc * frames;
302 if (lfo_phase > 6.2831853f) lfo_phase -= 6.2831853f;
303 // 2^(cents/1200) ~ 1 + cents * 0.000577623 for small values
304 step = base_step * pitch_factor * (1.0f + depth * 0.000577623f * sinf(lfo_phase));
305 }

308 template <bool ORGAN>
309 void renderBlock(float* out_left, float* out_right, int frames) {
310 updateVibrato(frames);
311 float amp = amplitude();
312 advanceEnvelopes(frames);
313 float amp_inc = (amplitude() - amp) / frames;
314
315 // local copies: the compiler can keep them in registers (the output
316 // buffers could otherwise alias the members)
317 float f = frac, st = step, lp = lp_state, lc = lp_coef, l = left, r = right;
318 float h0 = hist[0], h1 = hist[1], h2 = hist[2], h3 = hist[3];
319 for (int j = 0; j < frames; j++) {
320 // 4 point, 3rd order hermite interpolation
321 float c1 = 0.5f * (h2 - h0);
322 float c2 = h0 - 2.5f * h1 + 2.0f * h2 - 0.5f * h3;
323 float c3 = 0.5f * (h3 - h0) + 1.5f * (h1 - h2);
324 float sample = ((c3 * f + c2) * f + c1) * f + h1;
325
326 lp += lc * (sample - lp);
327 float out = lp * amp;
328 amp += amp_inc;
329 out_left[j] += out * l;
330 out_right[j] += out * r;
331
332 f += st;
333 while (f >= 1.0f) {
334 f -= 1.0f;
335 h0 = h1;
336 h1 = h2;
337 h2 = h3;
338 h3 = nextSample<ORGAN>();
339 }
340 }
341 frac = f;
342 lp_state = lp;
343 hist[0] = h0;
344 hist[1] = h1;
345 hist[2] = h2;
346 hist[3] = h3;
347 elapsed += frames;
348
349 // end: silent or no more data
350 if (decay_env * release_env < 0.0001f || ended > 4) state = Idle;
351 }

354 template <bool ORGAN>
355 inline float nextSample() {
356 if (ORGAN) return organ_source.next();
357 const SampleZone& z = *p_zone;
358 if (use_psola) {
359 float s = psola.next();
360 if (!decay_active && psola.time() >= z.decayStart) decay_active = true;
361 if (psola.finished()) ended++;
362 return s;
363 }
364 if (index >= z.loopEnd && z.isLooped()) {
365 index = z.loopStart;
366 decoder.seek(z.loopStart, z.loopPredictor, z.loopIndex);
367 }
368 if (index >= z.length) {
369 ended++;
370 return 0.0f;
371 }
372 if (index == z.decayStart) decay_active = true;
373 index++;
374 return decoder.next() * (1.0f / 32768.0f);
375 }
