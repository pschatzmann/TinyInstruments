222 template <class STREAM>
223 static bool readFully(STREAM& in, uint8_t* data, size_t len) {
224 size_t done = 0;
225 int retries = 0;
226 while (done < len) {
227 // char* is supported by all Arduino streams (e.g. fs::File hides the uint8_t* variant)
228 size_t n = in.readBytes((char*)(data + done), len - done);
229 if (n == 0) {
230 if (++retries > 10) return false;
231 continue;
232 }
233 retries = 0;
234 done += n;
235 }
236 return true;
237 }
238
239 static uint16_t u16(const uint8_t* p) { return (uint16_t)(p[0] | (p[1] << 8)); }
240 static uint32_t u32(const uint8_t* p) {
241 return (uint32_t)p[0] | ((uint32_t)p[1] << 8) | ((uint32_t)p[2] << 16) | ((uint32_t)p[3] << 24);
242 }
