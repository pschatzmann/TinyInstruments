188 void begin(const TonewheelOrgan* organ, uint8_t note, float sampleRate, bool percussion, uint32_t seed) {
189 p_organ = organ;
190 rate = sampleRate;
191 random = seed * 2654435761u + 1u;
192 table = TonewheelOrgan::sineTable();
193 for (int j = 0; j < TonewheelOrgan::DRAWBARS; j++) {
194 inc[j] = increment(note + TonewheelOrgan::offset(j));
195 phase[j] = nextRandom();
196 }
197 perc_inc = increment(note + organ->percussionOffset());
198 perc_phase = nextRandom();
199 perc_env = percussion ? organ->percussionGain() : 0.0f;
200 perc_mul = expf(-6.9077553f / (organ->percussionDecay() * sampleRate));
201 // key click: a few ms of noise, louder for higher registrations
202 click_env = organ->keyClick() * 0.15f;
203 click_mul = expf(-1.0f / (0.0015f * sampleRate));
204 click_lp = 0.0f;
205 version = organ->changes() + 1; // read the gains
206 counter = 0;
207 updateGains(true);
208 }
209
211 inline float next() {
212 if (counter == 0) {
213 updateGains(false);
214 counter = UPDATE;
215 }
216 counter--;
217 float out = 0.0f;
218 for (int k = 0; k < active_count; k++) {
219 int j = active[k];
220 out += gains[j] * sine(phase[j]);
221 gains[j] += steps[j];
222 phase[j] += inc[j];
223 }
224 if (perc_env > 0.00001f) {
225 out += perc_env * sine(perc_phase);
226 perc_phase += perc_inc;
227 perc_env *= perc_mul;
228 }
229 if (click_env > 0.00001f) {
230 // noise with a gentle lowpass
231 float noise = (float)(int32_t)nextRandom() * (1.0f / 2147483648.0f);
232 click_lp += 0.5f * (noise - click_lp);
233 out += click_env * click_lp;
234 click_env *= click_mul;
235 }
236 return out;
237 }

239 protected:
240 static constexpr int UPDATE = 32; // samples between the updates of the gains
241 const TonewheelOrgan* p_organ = nullptr;
242 const float* table = nullptr;
243 float rate = 44100.0f;

258 inline float sine(uint32_t p) const {
259 uint32_t index = p >> 22;
260 float frac = (float)(p & 0x3FFFFF) * (1.0f / 4194304.0f);
261 float a = table[index];
262 return a + (table[index + 1] - a) * frac;
263 }

268 }

272 uint32_t increment(int note) const {
273 float freq = 440.0f * powf(2.0f, (note - 69) / 12.0f);
274 while (freq > 5920.0f) freq *= 0.5f;
275 while (freq < 32.7f) freq *= 2.0f;
276 double inc = (double)freq / rate * 4294967296.0;
277 return (uint32_t)inc;
278 }
279
282 void updateGains(bool immediate) {
283 uint32_t v = p_organ->changes();
284 bool changed = v != version;
285 version = v;
286 active_count = 0;
287 for (int j = 0; j < TonewheelOrgan::DRAWBARS; j++) {
288 float target = p_organ->gain(j);
289 if (immediate) {
290 gains[j] = target;
291 steps[j] = 0.0f;
292 } else if (changed || steps[j] != 0.0f) {
293 steps[j] = (target - gains[j]) / UPDATE;
294 if (fabsf(steps[j]) < 1e-9f) {
295 steps[j] = 0.0f;
296 gains[j] = target;
297 }
298 }
299 if (gains[j] != 0.0f || target != 0.0f) active[active_count++] = (uint8_t)j;
300 }
