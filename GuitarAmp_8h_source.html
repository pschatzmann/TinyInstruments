100 void update() {
101 // 0..48 dB input gain
102 pre_gain = powf(10.0f, settings.drive * 48.0f / 20.0f) * 2.0f;
103 // the clipping limits the level: compensate the gain for low drive
104 post_gain = settings.level / (pre_gain < 1.0f ? 1.0f : sqrtf(pre_gain));
105 bias = settings.asymmetry;
106 bias_offset = shape(bias);
107 if (rate > 0) designFilters();
108 if (settings.echo > 0.0f && rate > 0) allocateEcho();
109 }

111 void begin(uint32_t sampleRate) override {
112 rate = sampleRate;
113 designFilters();
114 chorus_pos = 0;
115 memset(chorus_buffer, 0, sizeof(chorus_buffer));
116 lfo_phase = 0;
117 lfo_inc = 6.2831853f * 0.8f / rate;
118 chorus_base = 0.012f * rate; // 12 ms
119 chorus_depth = 0.003f * rate; // +/- 3 ms
120 freeEcho();
121 if (settings.echo > 0.0f) allocateEcho();
122 }

124 void process(float* left, float* right, int frames) override {
125 // the chorus lfo is slow: calculate it per block, interpolate linearly
126 float mod = 0.0f, mod_inc = 0.0f;
127 if (settings.chorus > 0.0f) {
128 mod = sinf(lfo_phase);
129 lfo_phase += lfo_inc * frames;
130 if (lfo_phase > 6.2831853f) lfo_phase -= 6.2831853f;
131 mod_inc = (sinf(lfo_phase) - mod) / frames;
132 }
133 for (int j = 0; j < frames; j++) {
134 // mono input
135 float x = (left[j] + right[j]) * 0.5f * pre_gain;
136 x = input_hp.process(x);
137
138 // 2x oversampled clipping: reduces the aliasing of the distortion
139 float mid_sample = 0.5f * (x + last_input);
140 last_input = x;
141 aa_lp.process(shape(mid_sample + bias) - bias_offset);
142 float y = aa_lp.process(shape(x + bias) - bias_offset);
143
144 // tone stack (bands at 0 dB are skipped) and speaker cabinet; the
145 // cabinet high pass also removes the DC of the asymmetric clipping
146 if (use_bass) y = bass_eq.process(y);
147 if (use_mid) y = mid_eq.process(y);
148 if (use_treble) y = treble_eq.process(y);
149 y = cab_hp.process(y);
150 y = cab_peak.process(y);
151 y = cab_lp1.process(y);
152 y = cab_lp2.process(y) * post_gain;
153
154 float out_l = y, out_r = y;
155
156 // echo (mono, with feedback)
157 if (echo_buffer != nullptr && settings.echo > 0.0f) {
158 float delayed = echo_buffer[echo_pos] * (1.0f / 32767.0f);
159 float fb = y + delayed * settings.echoFeedback;
160 if (fb > 1.0f) fb = 1.0f;
161 if (fb < -1.0f) fb = -1.0f;
162 echo_buffer[echo_pos] = (int16_t)(fb * 32767.0f);
163 if (++echo_pos >= echo_len) echo_pos = 0;
164 out_l += delayed * settings.echo;
165 out_r += delayed * settings.echo;
166 }
167
168 // stereo chorus: two modulated delays
169 if (settings.chorus > 0.0f) {
170 chorus_buffer[chorus_pos] = y;
171 float dl = chorusTap(chorus_base + chorus_depth * mod);
172 float dr = chorusTap(chorus_base - chorus_depth * mod);
173 mod += mod_inc;
174 out_l += settings.chorus * dl;
175 out_r += settings.chorus * dr;
176 chorus_pos = (chorus_pos + 1) & (CHORUS_SIZE - 1);
177 }
178
179 left[j] = out_l;
180 right[j] = out_r;
181 }
182 }

214 static inline float shape(float x) {
215 if (x > 3.0f) return 1.0f;
216 if (x < -3.0f) return -1.0f;
217 float x2 = x * x;
218 return x * (27.0f + x2) / (27.0f + 9.0f * x2);
219 }

236 void designFilters() {
237 float r = rate;
238 // a tight low end for high drive, but not above the cabinet low cut
239 float input_cut = settings.drive > 0.3f ? 120.0f : 60.0f;
240 if (input_cut > settings.lowCut) input_cut = settings.lowCut;
241 input_hp.highPass(input_cut, 0.7f, r);
242 aa_lp.lowPass(0.42f * r, 0.7f, 2.0f * r); // runs at 2x the rate
243 use_bass = settings.bass != 0.0f;
244 use_mid = settings.mid != 0.0f;
245 use_treble = settings.treble != 0.0f;
246 bass_eq.lowShelf(150.0f, settings.bass, r);
247 mid_eq.peak(800.0f, 0.8f, settings.mid, r);
248 treble_eq.highShelf(3000.0f, settings.treble, r);
249 // speaker cabinet: low cut, presence peak, steep high cut
250 cab_hp.highPass(settings.lowCut, 0.8f, r);
251 cab_peak.peak(2200.0f, 1.2f, settings.presence, r);
252 cab_lp1.lowPass(settings.highCut, 0.7f, r);
253 cab_lp2.lowPass(settings.highCut, 0.7f, r);
254 }

257 inline float chorusTap(float delay) {
258 int whole = (int)delay;
259 float frac = delay - whole;
260 int i0 = (chorus_pos - whole) & (CHORUS_SIZE - 1);
261 int i1 = (i0 - 1) & (CHORUS_SIZE - 1); // one sample older
262 float a = chorus_buffer[i0];
263 return a + (chorus_buffer[i1] - a) * frac;
264 }

266 void allocateEcho() {
267 int len = (int)(settings.echoTime * rate);
268 if (len < 1) len = 1;
269 if (len > (int)(0.5f * rate)) len = (int)(0.5f * rate);
270 if (echo_buffer != nullptr && len == echo_len) return;
271 freeEcho();
272 // 16 bit, in PSRAM if available
273 echo_buffer = (int16_t*)SampleMemory::allocate(len * sizeof(int16_t));
274 if (echo_buffer == nullptr) return;
275 memset(echo_buffer, 0, len * sizeof(int16_t));
276 echo_len = len;
277 echo_pos = 0;
278 }
279
280 void freeEcho() {
281 if (echo_buffer) SampleMemory::release(echo_buffer);
282 echo_buffer = nullptr;
283 echo_len = 0;
284 }
