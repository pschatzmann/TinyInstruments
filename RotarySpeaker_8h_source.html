81 void process(float* left, float* right, int frames) override {

84 drum.update(is_stopped ? 0.0f : (is_fast ? drum.fast : drum.slow), frames);

86 const float base = 0.001f * rate; // delay in the middle of the modulation
87 float k = drive > 0.0f ? 1.0f + 12.0f * drive : 0.0f;
88 float makeup = k > 0.0f ? 1.0f / softClip(k * 0.5f) * 0.5f : 1.0f;
89
90 for (int j = 0; j < frames; j++) {

94 float lo = lp2.process(lp1.process(x));
95 float hi = hp2.process(hp1.process(x));
96 buffer_hi[pos] = hi;

102 float horn_am_l = 0.7f + 0.3f * horn.c;
103 float horn_am_r = 0.7f - 0.3f * horn.s;

111 right[j] = hr * horn_am_r + dr * drum_am_r;

114 drum.rotate();

142 void update(float target, int frames) {
143 float tau = target > freq ? up : down;
144 float k = 1.0f - expf(-(float)frames / (tau * rate));

155 }

161 }
